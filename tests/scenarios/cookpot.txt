// PLAYER: DST Crock Pot (cooking.lua, preparedfoods.lua, stewer.lua, cookpot.lua): the resolver
// (tags, priorities, ties, wet goop, cooked and aliased names, items that do not exist yet), the
// pot block (fill, cook over DS time, finished dish, harvest with freshness, spoiling into rot,
// eating, belly temperature), hammering (loot, loaded ingredients) and burning.
#wait-player {P} 180
gamemode survival {P}
ds hound pause on
ds codex {P}
#expect read the Codex Umbra
#sleep 3
ds clock set 1 day 100
ds island current
execute in dontstarve:islands run fill {X+5} 76 {Z-18} {X+45} 96 {Z+18} minecraft:air
execute in dontstarve:islands run fill {X+5} 75 {Z-18} {X+45} 75 {Z+18} minecraft:stone
// Loose worldgen flint that fell onto the platform would mix into the item checks.
execute in dontstarve:islands run kill @e[type=item,x={X+5},y=70,z={Z-18},dx=41,dy=27,dz=37]
execute in dontstarve:islands run tp {P} {X+22} 76 {Z-10} 0 30
#sleep 2
ds island current
// ---- The recipe is in the DS book (cutstone 3, charcoal 6, twigs 6, SCIENCE_ONE, placer).
ds check dsrecipe cookpot
#expect OK dsrecipe cookpot tech \{SCIENCE=1\} placer
// ---- Resolver (cooking.lua GetCandidateRecipes: only the top priority group, CalculateRecipe picks
// among them by weight). Replies: product, its priority and cook time, candidates, summed tags.
// meatballs (priority -1) beats nothing else here; wet goop is -10.
ds cook calc meat berries berries berries
#expect cook calc meatballs priority -1 cooktime 0\.75 candidates meatballs tags fruit=1\.5/meat=1
// Items by registry id work like prefab names: 4 berries make jam (priority 0, fruit 2).
ds cook calc dontstarve:berries dontstarve:berries dontstarve:berries dontstarve:berries
#expect cook calc jammypreserves priority 0 cooktime 0\.5 candidates jammypreserves tags fruit=2
ds cook calc carrot carrot carrot carrot
#expect cook calc ratatouille priority 0 cooktime 1 candidates ratatouille tags veggie=4
// Priority: bonestew (0, meat >= 3) beats meatballs (-1).
ds cook calc meat meat meat meat
#expect cook calc bonestew priority 0 cooktime 0\.75 candidates bonestew tags meat=4
// kabobs (5): meat and twigs, at most 1 inedible.
ds cook calc meat twigs berries berries
#expect cook calc kabobs priority 5 cooktime 2 candidates kabobs tags fruit=1/inedible=1/meat=1
// Two twigs too many make no kabobs; monster meat x2 rules out the kabobs and the lasagna (twigs):
// the fallback is wet goop (priority -10).
ds cook calc monstermeat monstermeat berries twigs
#expect cook calc wetgoop priority -10 cooktime 0\.25 candidates wetgoop
// A tie in the top priority group (hotchili and turkeydinner, both 10): both are candidates.
ds cook calc drumstick drumstick meat carrot carrot
#expect cook calc (hotchili|turkeydinner) priority 10 cooktime (0\.5|3) candidates hotchili/turkeydinner
// Cooked names carry the precook tag; the alias cookedmonstermeat is monstermeat_cooked.
ds cook calc butter berries_cooked egg
#expect cook calc waffles priority 10 cooktime 0\.5 candidates waffles tags dairy=1/egg=1/fat=1/fruit=0\.5/precook=1
ds cook calc cookedmonstermeat cookedmonstermeat cookedmonstermeat berries
#expect cook calc monsterlasagna priority 10 cooktime 0\.5 candidates monsterlasagna tags fruit=0\.5/meat=3/monster=3/precook=3
// Mod items: twigs only is beefalo feed (-5); vanilla flint is no ingredient and adds nothing.
ds cook calc dontstarve:twigs dontstarve:twigs dontstarve:twigs dontstarve:twigs
#expect cook calc beefalofeed priority -5 cooktime 0\.5 candidates beefalofeed tags inedible=4
ds cook calc minecraft:flint berries
#expect cook calc jammypreserves priority 0 cooktime 0\.5 candidates jammypreserves tags fruit=0\.5
// Dried variants of meat count as meat too (jerky: meat_dried is its own prefab with the dried tag).
ds cook calc meat_dried meat_dried meat_dried meat_dried
#expect cook calc bonestew .* tags dried=4/meat=4
// ---- Prepared food items: DST edible values (meatballs CALORIES_SMALL*5 = 62.5 hunger), perishable
// into rot, wet goop is plain nothing.
item replace entity {P} weapon.mainhand with dontstarve:meatballs 1
ds perish {P}
#expect Meatballs x1 fresh 100\.0%
ds stats {P} set hunger 50
ds eat {P}
#expect hunger 112\.[45]\d
item replace entity {P} weapon.mainhand with dontstarve:wetgoop 1
ds perish {P}
#expect Wet Goop x1 fresh 100\.0%
ds stats {P} set hunger 50
ds eat {P}
#expect hunger (49\.9\d|50\.0\d)
clear {P}

#expect cookpot put berries slot 1

#expect cookpot put berries slot 2

#expect cookpot put berries slot 3

#expect cookpot cannot put berries full or busy
ds cookpot {X+10} 76 {Z+4} check
#expect cookpot state empty burnt false items 4

ds cookpot {X+10} 76 {Z+4} cook
#expect cookpot cooking jammypreserves ticks 500
ds cookpot {X+10} 76 {Z+4} check

#expect cookpot cannot cook busy

// 2 monster meat, no twigs: monsterlasagna (PERISH_FAST 2880 s). Ingredients at 1 % freshness: the dish
// starts at 1 - (1 - 0.01) * 0.5 = 0.505 of its perish time (stewer.lua StartCooking).
ds cookpot {X+18} 76 {Z+4} cook
#expect cookpot cooking monsterlasagna ticks 500
ds cookpot {X+18} 76 {Z+4} check
#expect product monsterlasagna .* spoilage 0\.505
// stewer:LongUpdate: 20 DS s finish the cooking and start the spoil timer (2880 x 0.505 = 1454.4 s = 72720 ticks).

#expect cookpot state done .* product monsterlasagna cookticks 0 spoilticks 72[12]\d\d spoilage 0\.505
// Half of that time later the dish is 0.505 x 0.49 = 24.9 % fresh: stale.
ds cookpot {X+18} 76 {Z+4} advance 727.2
clear {P}
ds cookpot {X+18} 76 {Z+4} harvest {P}
#expect cookpot harvested monsterlasagna x1 fresh 2[45]\.\d

#expect x1 stale 2[45]\.\d%
