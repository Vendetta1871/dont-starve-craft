// Ghost perks and the all-ghost wipe countdown (needs a real online player).
// NOTE: this scenario leaves the player where the wipe logic put them.
#wait-player {P} 180
ds island next
ds island current
#expect Current island
// Send them to the islands and turn them into a ghost.
execute in dontstarve:islands run tp {P} {X} 80 {Z}
#sleep 3
// Only members of the current run count for the wipe (reading the Codex Umbra sets this).
ds run {P} join
#expect joined run
ds ghost {P} check
#expect ghost=false dimension=dontstarve:islands
ds ghost {P} on
#expect is now a ghost
data get entity {P} abilities
#expect mayfly: 1
data get entity {P} active_effects
#expect minecraft:invisibility
ds ghost {P} check
#expect ghost=true dimension=dontstarve:islands
// All island players are ghosts now: the collapse countdown starts.
#expect-log All survivors have perished 60
// A living soul walks the island again: the countdown is averted.
ds ghost {P} off
#expect-log A living soul walks the island again 60
ds ghost {P} check
#expect ghost=false dimension=dontstarve:islands
