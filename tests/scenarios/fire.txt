// PLAYER: DST fire (burnable.lua, propagator.lua, fueled.lua, campfire.lua, firepit.lua,
// standardcomponents.lua): the campfire burns down its fuel in four sections and leaves ash,
// the fire pit stays, fuel is added by item, fires give light and warmth, a burning
// prototyper heats its neighbours up until they smoulder and catch fire, hurts creatures,
// ends as a burnt wreck, and hammering works on burning and burnt structures.
#wait-player {P} 180
gamemode survival {P}
ds hound pause on
ds codex {P}
#expect read the Codex Umbra
#sleep 3
ds clock set 1 night 500
ds island current
// A wide clearing: fire range reaches 8 blocks, natural plants past the edge would catch fire.
execute in dontstarve:islands run fill {X+8} 76 {Z-18} {X+42} 96 {Z+18} minecraft:air
execute in dontstarve:islands run fill {X+8} 75 {Z-18} {X+42} 75 {Z+18} minecraft:stone
// Loose worldgen flint that fell onto the platform would mix into the item checks.
execute in dontstarve:islands run kill @e[type=item,x={X+8},y=70,z={Z-18},dx=34,dy=27,dz=36]
execute in dontstarve:islands run tp {P} {X+25} 76 {Z} 0 30
#sleep 2
ds island current
// ---- Campfire: built through the crafting menu (3 cut grass + 2 logs), starts at CAMPFIRE_FUEL_START
// (135 DS s = section 3) and burns at once; light level follows the fuel section.
clear {P}
give {P} dontstarve:cutgrass 3
give {P} minecraft:spruce_log 2
ds craft {P} campfire
#expect buffered \[campfire\]
ds place {P} campfire {X+25} 76 {Z+2}
#expect placed campfire
#sleep 1
ds fire check {X+25} 76 {Z+2}
#expect fire campfire burning heat 0\.0 paused false ticksleft -1 warmth \d+ lightblock false fuel 13[45]\.0 section [23] light [45] queued false
// At night the campfire lights the player two blocks away (block light 5 - 2 = 3 -> 0.20).
execute in dontstarve:islands run tp {P} {X+25} 76 {Z+1} 0 30
ds island current
#sleep 1
ds stats {P}
#expect light 0\.(1[3-9]|2\d)
// ---- Warmth (heater: heat 85..100 x (1 - d^2/100)): in a winter night body temperature rises
// 1 degree per DS s next to the fire (without it, stats.txt: it falls 1 per DS s).
ds clock set 28 night 500
ds stats {P} set temperature 2
#ds-wait 8
ds stats {P}
#expect temperature (9\.[6-9]|10\.[0-4])\d
ds clock set 1 day 100
// ---- Fuel sections (fueled.lua: sections = 4 over CAMPFIRE_FUEL_MAX 270): <67.5 / <135 / <202.5 / more.
ds fire fuel {X+25} 76 {Z+2} set 140
#expect fuel 140\.0 section 3 light 5
#ds-wait 10
ds fire check {X+25} 76 {Z+2}
#expect fuel 1(29|30|31)\.\d section 2
ds fire fuel {X+25} 76 {Z+2} set 70
#expect section 2
#ds-wait 5
ds fire check {X+25} 76 {Z+2}
#expect fuel 6[4-6]\.\d section 1 light [34]
// ---- Fuel items (fuel component): log MED_FUEL 45, cut grass SMALL_FUEL 15, rocks are no fuel.
ds fire fuel {X+25} 76 {Z+2} set 100
item replace entity {P} weapon.mainhand with minecraft:spruce_log 3
ds fire feed {P} {X+25} 76 {Z+2}
#expect fuel 14[45]\.\d section 3 .* left 2
item replace entity {P} weapon.mainhand with dontstarve:cutgrass 2
ds fire feed {P} {X+25} 76 {Z+2}
#expect fuel (159|160)\.\d section 3 .* left 1
item replace entity {P} weapon.mainhand with dontstarve:rocks 2
ds fire feed {P} {X+25} 76 {Z+2}
#expect not fuel for campfire
// The campfire cannot be hammered or lit by a torch (it is lit while it has fuel).
item replace entity {P} weapon.mainhand with dontstarve:hammer
ds work {P} {X+25} 76 {Z+2}
#expect not workable dontstarve:campfire
ds fire light {P} {X+25} 76 {Z+2}
#expect cannot light campfire
// ---- Charcoal is queued once the top section is reached; a campfire that burns out is gone
// and leaves charcoal (when that item exists) or ash.
// Stand clear of the drops: a survival player picks up items that land next to them.
execute in dontstarve:islands run tp {P} {X+25} 76 {Z-6} 0 30
ds fire fuel {X+25} 76 {Z+2} set 220
#expect section 4 .* queued true
ds fire fuel {X+25} 76 {Z+2} set 2
#ds-wait 3
ds fire check {X+25} 76 {Z+2}
#expect fire air not burnable
execute in dontstarve:islands run kill @e[type=item,x={X+25},y=76,z={Z+2},distance=..4]
#expect Killed (Charcoal|Ashes)
// A campfire that never reached the top section leaves plain ash.
execute in dontstarve:islands run setblock {X+22} 76 {Z+2} dontstarve:campfire
ds island current
#sleep 1
ds fire check {X+22} 76 {Z+2}
#expect fire campfire burning .* fuel 13[45]\.0 section [23] .* queued false
ds fire fuel {X+22} 76 {Z+2} set 1
#ds-wait 3
ds fire check {X+22} 76 {Z+2}
#expect fire air not burnable
execute in dontstarve:islands run kill @e[type=item,x={X+22},y=76,z={Z+2},distance=..4]
#expect Killed Ashes
// ---- Fire pit (2 logs + 12 rocks): FIREPIT_FUEL_START 180 = section 3, fuel x FIREPIT_BONUS_MULT 2.
clear {P}
give {P} minecraft:spruce_log 2
give {P} dontstarve:rocks 12
ds craft {P} firepit
#expect buffered \[firepit\]
execute in dontstarve:islands run tp {P} {X+25} 76 {Z+1} 0 30
ds place {P} firepit {X+28} 76 {Z+2}
#expect placed firepit
#sleep 1
ds fire check {X+28} 76 {Z+2}
#expect fire firepit burning .* fuel 1(79|80)\.\d section 3 light 5 queued false
// It stays when the fuel is gone (and goes cold) ...
ds fire fuel {X+28} 76 {Z+2} set 3
#ds-wait 5
ds fire check {X+28} 76 {Z+2}
#expect fire firepit idle heat 0\.0 paused false fuel 0\.0 section 0 light 0 queued false
// ... and burns again with new fuel: a log is worth 45 x 2.
item replace entity {P} weapon.mainhand with minecraft:spruce_log 2
ds fire feed {P} {X+28} 76 {Z+2}
#expect fire firepit burning .* fuel 90\.0 section 2 .* left 1
ds fire fuel {X+28} 76 {Z+2} set 300
#expect section 4 .* queued true
// Hammer: 4 hits, then half the recipe (1 log, 6 rocks) and a heap of ash.
item replace entity {P} weapon.mainhand with dontstarve:hammer
ds work {P} {X+28} 76 {Z+2} 3
#expect worked dontstarve:firepit hits 3 left 1
ds work {P} {X+28} 76 {Z+2}
#expect finished dontstarve:firepit hits 1
execute in dontstarve:islands run kill @e[type=item,nbt={Item:{id:"minecraft:spruce_log",count:1}},x={X+28},y=76,z={Z+2},distance=..4]
#expect Killed Spruce Log
execute in dontstarve:islands run kill @e[type=item,nbt={Item:{id:"dontstarve:rocks",count:6}},x={X+28},y=76,z={Z+2},distance=..4]
#expect Killed Rocks
execute in dontstarve:islands run kill @e[type=item,nbt={Item:{id:"dontstarve:ash"}},x={X+28},y=76,z={Z+2},distance=..4]
#expect Killed Ashes
// ---- Fire spread between prototypers (MakeLargeBurnable + MakeLargePropagator: burn 30 DS s,
// flashpoint 45-55, range 6-8, heat 6-9.5/s, damage range 3). A burns, a pig beside it is hurt,
// B three blocks away heats up past its flashpoint, smoulders and ignites later.
ds clock set 2 day 100
execute in dontstarve:islands run setblock {X+19} 76 {Z-4} dontstarve:researchlab
ds island current
execute in dontstarve:islands run setblock {X+22} 76 {Z-4} dontstarve:researchlab
ds island current
execute in dontstarve:islands run summon minecraft:pig {X+20} 76 {Z-4} {NoAI:1b,PersistenceRequired:1b}
ds fire check {X+19} 76 {Z-4}
#expect fire researchlab idle heat 0\.0 paused false burnt false
ds fire ignite {X+19} 76 {Z-4}
#expect ignited researchlab
ds fire check {X+19} 76 {Z-4}
#expect fire researchlab burning heat 0\.0 paused false ticksleft (14[5-9]\d|1500) warmth 180 lightblock true
execute in dontstarve:islands if block {X+19} 77 {Z-4} minecraft:light
#expect Test passed
// Fire damage starts after a second in the fire; after 6 DS s the pig is hurt but alive.
#ds-wait 6
execute in dontstarve:islands run data get entity @e[type=minecraft:pig,limit=1] Health
#expect : [0-9]\.\d+f
execute in dontstarve:islands run kill @e[type=minecraft:pig]
// After 20 DS s B has absorbed its flashpoint and smoulders (smoulder time >= flashpoint / 2.2).
#ds-wait 14
ds fire check {X+22} 76 {Z-4}
#expect fire researchlab smoldering heat [0-9.]+ paused true
// A burns out after 30 DS s: a burnt wreck (canlight false), its light is gone. (It may hold
// a little heat from B by now: a burnt structure stays burnable.)
#ds-wait 12
ds fire check {X+19} 76 {Z-4}
#expect fire researchlab idle heat [0-9.]+ paused (true|false) burnt true
execute in dontstarve:islands if block {X+19} 77 {Z-4} minecraft:light
#expect Test failed
// B has caught fire by 69 DS s at the latest (burning, or already burnt down); it stopped smouldering.
#ds-wait 43
ds fire check {X+22} 76 {Z-4}
#expect fire researchlab (burning|idle)
// Removing the blocks forgets their fire.
execute in dontstarve:islands run fill {X+18} 76 {Z-8} {X+24} 77 {Z-1} minecraft:air
ds island current
#ds-wait 2
ds fire check {X+22} 76 {Z-4}
#expect fire air not burnable
execute in dontstarve:islands if block {X+22} 77 {Z-4} minecraft:light
#expect Test failed

#expect finished dontstarve:researchlab hits 1
