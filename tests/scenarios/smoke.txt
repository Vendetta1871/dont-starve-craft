// Boot, registration, and command wiring.
ds island current
#expect Current island \d+ centered at 0, 0, 0
ds check block dontstarve:marble
#expect OK block
ds check block dontstarve:grave
#expect OK block
ds check block dontstarve:resurrectionstatue
#expect OK block
ds check block dontstarve:sky_portal
#expect OK block
ds check item dontstarve:moonrocknugget
#expect OK item
ds check item dontstarve:moonrockidol
#expect OK item
ds check item dontstarve:reviver
#expect OK item
ds check item dontstarve:cutgrass
#expect OK item
// DS recipes come from the DST data through the DS crafting menu.
ds check dsrecipe reviver
#expect OK dsrecipe reviver tech \{\} filters \[RESTORATION\]
ds check dsrecipe resurrectionstatue
#expect OK dsrecipe resurrectionstatue tech \{MAGIC=2\} placer
ds check dsrecipe researchlab
#expect OK dsrecipe researchlab tech \{\} placer filters \[PROTOTYPERS, STRUCTURES\]
ds check dsrecipe researchlab2
#expect OK dsrecipe researchlab2 tech \{SCIENCE=1\} placer
ds check block dontstarve:researchlab3
#expect OK block
// Recipes for content not ported yet stay out of the book: the Prestihatitator
// (rabbit, top hat) and the Moon Rock Idol (purple gem, at a Celestial Orb).
ds check dsrecipe researchlab4
#expect MISSING dsrecipe
ds check dsrecipe moonrockidol
#expect MISSING dsrecipe
// The vanilla torch recipe is vanilla again: the grid is off on the islands anyway.
ds check recipe minecraft:torch
#expect OK recipe
ds check tag minecraft:mineable/shovel
#expect OK tag
// Negative controls: the checker must actually look things up.
ds check item dontstarve:not_a_real_item
#expect MISSING item
ds check recipe dontstarve:not_a_real_recipe
#expect MISSING recipe
